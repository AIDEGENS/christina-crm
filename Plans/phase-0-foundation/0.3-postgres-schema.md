## Objective
Deploy the `crm` Postgres schema with all 9 tables, row-level security on `tenant_id`, required indexes, and extensions (pgcrypto, pg_trgm, uuid-ossp). Seed 2 tenants.

## Approach

### Install Drizzle ORM

```bash
cd packages/db
pnpm add drizzle-orm postgres
pnpm add -D drizzle-kit @types/node
```

### Create drizzle config

`packages/db/drizzle.config.ts`:
```typescript
import type { Config } from "drizzle-kit";

export default {
  schema: "./src/schema/*",
  out: "./migrations",
  dialect: "postgresql",
  dbCredentials: { url: process.env.DATABASE_URL! },
  schemaFilter: ["crm"],
} satisfies Config;
```

### Schema files

Create `packages/db/src/schema/` with one file per table. Use `pgSchema('crm')` so everything lives in the `crm` schema.

`packages/db/src/schema/tenants.ts`:
```typescript
import { pgSchema, uuid, text, timestamp, jsonb } from "drizzle-orm/pg-core";

export const crm = pgSchema("crm");

export const tenants = crm.table("tenants", {
  id: uuid("id").defaultRandom().primaryKey(),
  name: text("name").notNull(),
  npi: text("npi"),
  ein: text("ein"),  // encrypt at app layer with pgcrypto
  medicarePtan: text("medicare_ptan"),
  macJurisdiction: text("mac_jurisdiction"),  // "noridian_je"
  mediCalProviderId: text("medi_cal_provider_id"),
  serviceArea: text("service_area").array(),
  stripeCustomerId: text("stripe_customer_id"),
  metadata: jsonb("metadata").default({}),
  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow().notNull(),
});
```

Repeat for: `users`, `organizations`, `contacts`, `referrals`, `referralNotes`, `referralDocuments`, `bdVisits`, `auditLog`.

Follow exact column definitions in `../../REF/CRM-03-DATA-MODEL.md`. PHI columns (patient_dob, patient_mrn, patient_address, patient_member_id) use `text` with app-layer encryption via pgcrypto.

### Generate migration


This creates `migrations/0000_initial_crm_schema.sql`. Inspect it before applying.

### Add RLS policies (hand-written SQL migration)

Drizzle doesn't generate RLS. Add a migration file:

`packages/db/migrations/0001_rls_policies.sql`:
```sql
-- Enable RLS on all tenant-scoped tables
ALTER TABLE crm.referrals ENABLE ROW LEVEL SECURITY;
ALTER TABLE crm.organizations ENABLE ROW LEVEL SECURITY;
ALTER TABLE crm.contacts ENABLE ROW LEVEL SECURITY;
ALTER TABLE crm.referral_notes ENABLE ROW LEVEL SECURITY;
ALTER TABLE crm.referral_documents ENABLE ROW LEVEL SECURITY;
ALTER TABLE crm.bd_visits ENABLE ROW LEVEL SECURITY;
ALTER TABLE crm.users ENABLE ROW LEVEL SECURITY;
ALTER TABLE crm.audit_log ENABLE ROW LEVEL SECURITY;

-- Tenants table: users can only see their own tenant
ALTER TABLE crm.tenants ENABLE ROW LEVEL SECURITY;
CREATE POLICY tenant_self_select ON crm.tenants
  FOR SELECT USING (id = current_setting('app.current_tenant', true)::uuid);

-- Generic tenant_isolation policy for all child tables
CREATE POLICY tenant_isolation ON crm.referrals
  USING (tenant_id = current_setting('app.current_tenant', true)::uuid);

CREATE POLICY tenant_isolation ON crm.organizations
  USING (tenant_id = current_setting('app.current_tenant', true)::uuid);

CREATE POLICY tenant_isolation ON crm.contacts
  USING (tenant_id = current_setting('app.current_tenant', true)::uuid);

CREATE POLICY tenant_isolation ON crm.referral_notes
  USING (tenant_id = current_setting('app.current_tenant', true)::uuid);

CREATE POLICY tenant_isolation ON crm.referral_documents
  USING (tenant_id = current_setting('app.current_tenant', true)::uuid);

CREATE POLICY tenant_isolation ON crm.bd_visits
  USING (tenant_id = current_setting('app.current_tenant', true)::uuid);

CREATE POLICY tenant_isolation ON crm.users
  USING (tenant_id = current_setting('app.current_tenant', true)::uuid);

CREATE POLICY tenant_isolation ON crm.audit_log
  USING (tenant_id = current_setting('app.current_tenant', true)::uuid);

-- Audit log is append-only
CREATE POLICY audit_log_no_update ON crm.audit_log FOR UPDATE USING (false);
CREATE POLICY audit_log_no_delete ON crm.audit_log FOR DELETE USING (false);

-- Grant usage on schema to application role
CREATE ROLE crm_app NOLOGIN;
GRANT USAGE ON SCHEMA crm TO crm_app;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA crm TO crm_app;
GRANT SELECT, USAGE ON ALL SEQUENCES IN SCHEMA crm TO crm_app;

-- Application connects as a user that inherits crm_app
-- It sets app.current_tenant per request in middleware
```

### Add indexes (hand-written)

`packages/db/migrations/0002_indexes.sql`:
```sql
-- Pipeline queries (most frequent)
CREATE INDEX idx_referrals_tenant_status ON crm.referrals(tenant_id, status);
CREATE INDEX idx_referrals_tenant_created ON crm.referrals(tenant_id, created_at DESC);
CREATE INDEX idx_referrals_source_org ON crm.referrals(source_org_id);
CREATE INDEX idx_referrals_assigned ON crm.referrals(assigned_to) WHERE assigned_to IS NOT NULL;

-- Org scorecards (30-day window)
CREATE INDEX idx_referrals_org_recent ON crm.referrals(source_org_id, created_at DESC);

-- BD visits
CREATE INDEX idx_bd_visits_user_date ON crm.bd_visits(user_id, visit_date DESC);
CREATE INDEX idx_bd_visits_org_date ON crm.bd_visits(org_id, visit_date DESC);

-- Audit log (time-series)
CREATE INDEX idx_audit_tenant_time ON crm.audit_log(tenant_id, created_at DESC);

-- Full-text / fuzzy search (Phase 4)
CREATE EXTENSION IF NOT EXISTS pg_trgm;
CREATE INDEX idx_orgs_name_trgm ON crm.organizations USING gin(name gin_trgm_ops);
CREATE INDEX idx_contacts_name_trgm ON crm.contacts USING gin((first_name || ' ' || last_name) gin_trgm_ops);
```

### Apply migrations

```bash
# Connect as master user (admin role needs to create extensions + RLS)
export DATABASE_URL=postgres://crm_admin:<password>@<endpoint>/crm

# Install pgcrypto + uuid extensions first
psql $DATABASE_URL -c "CREATE EXTENSION IF NOT EXISTS pgcrypto;"
psql $DATABASE_URL -c "CREATE EXTENSION IF NOT EXISTS \"uuid-ossp\";"

# Apply migrations
cd packages/db
pnpm drizzle-kit push:pg    # applies schema
psql $DATABASE_URL -f migrations/0001_rls_policies.sql
psql $DATABASE_URL -f migrations/0002_indexes.sql
```

## Acceptance criteria

- [ ] All 9 tables exist in `crm` schema (verify with `\dt crm.*` in psql)
- [ ] RLS is enabled on all 9 tables (verify with `SELECT tablename, rowsecurity FROM pg_tables WHERE schemaname='crm';`)
- [ ] All indexes created (verify with `\di crm.*`)
- [ ] pgcrypto + uuid-ossp + pg_trgm extensions installed
- [ ] 2 tenants seeded (HH + Hospice)
- [ ] Connecting as a role without `app.current_tenant` set returns 0 rows from `crm.referrals` (RLS blocks)
- [ ] Setting `app.current_tenant` to a tenant UUID returns only that tenant's rows
- [ ] `audit_log` cannot be UPDATEd or DELETEd (RLS blocks)

## Common issues

- **RLS policy doesn't apply** — make sure you're NOT connected as the superuser; RLS is bypassed for BYPASSRLS roles. App should connect as `crm_app` or similar.
- **"current setting \"app.current_tenant\" is missing"** — pass `true` as second arg to `current_setting()` so it returns NULL instead of erroring
- **Drizzle can't see `crm` schema** — ensure `schemaFilter: ["crm"]` in drizzle.config.ts
- **Migration order matters** — schema first (0000), THEN RLS (0001), THEN indexes (0002)
