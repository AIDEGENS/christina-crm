## Objective
Provision the minimum AWS footprint needed for the MVP in **us-west-1 (N. California)**. Everything else (Fargate, full VPC, WAF) we add post-MVP. For the MVP demo, we need: VPC + RDS + S3 + KMS + SES + IAM.

## Approach

### Region and data residency

Client requires California data residency. Use `us-west-1` (N. California) for all resources. **Do NOT use us-west-2 (Oregon) or us-east-1 (Virginia).**

### Terraform structure

Create `infra/` at the monorepo root:

```
infra/
├── main.tf              # provider + backend
├── vpc.tf               # VPC, subnets, SG
├── rds.tf               # Postgres + KMS key
├── s3.tf                # buckets + KMS keys (per-tenant later)
├── ses.tf               # SES identity for outbound email
├── iam.tf               # IAM roles for ECS, Vercel, devs
├── variables.tf
├── outputs.tf
└── terraform.tfvars     # gitignored
```

### main.tf

```hcl
terraform {
  required_version = ">= 1.7"
  required_providers {
    aws = { source = "hashicorp/aws", version = "~> 5.0" }
  }
  backend "s3" {
    # Bootstrap this manually before first apply
    bucket = "medical-platform-tfstate-us-west-1"
    key    = "crm/terraform.tfstate"
    region = "us-west-1"
    encrypt = true
  }
}

provider "aws" {
  region = "us-west-1"
  default_tags {
    tags = {
      Project     = "medical-crm"
      Environment = var.env
      ManagedBy   = "terraform"
      Compliance  = "hipaa"
    }
  }
}
```

### vpc.tf — minimal VPC

```hcl
resource "aws_vpc" "main" {
  cidr_block           = "10.0.0.0/16"
  enable_dns_hostnames = true
  enable_dns_support   = true
  tags = { Name = "crm-vpc-${var.env}" }
}

# 2 private subnets (RDS requires 2 AZs)
resource "aws_subnet" "private" {
  count             = 2
  vpc_id            = aws_vpc.main.id
  cidr_block        = "10.0.${count.index + 1}.0/24"
  availability_zone = element(["us-west-1a", "us-west-1c"], count.index)
  tags = { Name = "crm-private-${count.index}" }
}

resource "aws_db_subnet_group" "main" {
  name       = "crm-db-subnets-${var.env}"
  subnet_ids = aws_subnet.private[*].id
}

resource "aws_security_group" "rds" {
  name   = "crm-rds-${var.env}"
  vpc_id = aws_vpc.main.id

  ingress {
    from_port   = 5432
    to_port     = 5432
    protocol    = "tcp"
    # DEV ONLY: allow from your IP. Production: restrict to VPC CIDR + Vercel IPs
    cidr_blocks = [var.dev_ingress_cidr]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}
```

### rds.tf — Postgres + KMS

```hcl
resource "aws_kms_key" "rds" {
  description             = "RDS encryption key for CRM"
  deletion_window_in_days = 30
  enable_key_rotation     = true
}

resource "aws_kms_alias" "rds" {
  name          = "alias/crm-rds-${var.env}"
  target_key_id = aws_kms_key.rds.key_id
}

resource "random_password" "db_master" {
  length  = 32
  special = true
}

resource "aws_db_instance" "crm" {
  identifier                 = "crm-${var.env}"
  engine                     = "postgres"
  engine_version             = "16.2"
  instance_class             = "db.t4g.small"   # MVP; upgrade in prod
  allocated_storage          = 20
  max_allocated_storage      = 100
  storage_type               = "gp3"
  storage_encrypted          = true
  kms_key_id                 = aws_kms_key.rds.arn

  db_name                    = "crm"
  username                   = "crm_admin"
  password                   = random_password.db_master.result

  db_subnet_group_name       = aws_db_subnet_group.main.name
  vpc_security_group_ids     = [aws_security_group.rds.id]
  publicly_accessible        = true   # DEV ONLY; false in prod

  backup_retention_period    = 7
  backup_window              = "09:00-10:00"  # UTC = 2am PST
  maintenance_window         = "Mon:11:00-Mon:12:00"

  deletion_protection        = true
  skip_final_snapshot        = false
  final_snapshot_identifier  = "crm-${var.env}-final-${formatdate("YYYYMMDD", timestamp())}"

  performance_insights_enabled = true
  enabled_cloudwatch_logs_exports = ["postgresql"]

  tags = { Name = "crm-rds-${var.env}" }
}

output "db_endpoint" { value = aws_db_instance.crm.endpoint }
output "db_password" { value = random_password.db_master.result; sensitive = true }
```

### s3.tf — KMS-encrypted bucket

```hcl
resource "aws_kms_key" "s3" {
  description             = "S3 encryption key for CRM"
  deletion_window_in_days = 30
  enable_key_rotation     = true
}

resource "aws_s3_bucket" "crm_docs" {
  bucket = "medical-crm-docs-${var.env}-us-west-1"
}

resource "aws_s3_bucket_server_side_encryption_configuration" "crm_docs" {
  bucket = aws_s3_bucket.crm_docs.id
  rule {
    apply_server_side_encryption_by_default {
      kms_master_key_id = aws_kms_key.s3.arn
      sse_algorithm     = "aws:kms"
    }
    bucket_key_enabled = true
  }
}

resource "aws_s3_bucket_public_access_block" "crm_docs" {
  bucket                  = aws_s3_bucket.crm_docs.id
  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}

resource "aws_s3_bucket_versioning" "crm_docs" {
  bucket = aws_s3_bucket.crm_docs.id
  versioning_configuration { status = "Enabled" }
}

output "s3_bucket" { value = aws_s3_bucket.crm_docs.id }
output "kms_key_arn" { value = aws_kms_key.s3.arn }
```

### Variables


### Bootstrap the Terraform backend bucket

Terraform needs an S3 bucket to store state. Create it manually first:

```bash
aws s3api create-bucket \
  --bucket medical-platform-tfstate-us-west-1 \
  --region us-west-1 \
  --create-bucket-configuration LocationConstraint=us-west-1

aws s3api put-bucket-versioning \
  --bucket medical-platform-tfstate-us-west-1 \
  --versioning-configuration Status=Enabled

aws s3api put-bucket-encryption \
  --bucket medical-platform-tfstate-us-west-1 \
  --server-side-encryption-configuration '{"Rules":[{"ApplyServerSideEncryptionByDefault":{"SSEAlgorithm":"AES256"}}]}'
```

### Apply

```bash
cd infra
terraform init
terraform plan -out=tfplan
terraform apply tfplan
```

- [ ] Can connect to RDS from local with `psql` using the outputted endpoint + password
- [ ] Terraform state file in s3://medical-platform-tfstate-us-west-1/
- [ ] `terraform.tfvars` is gitignored
- [ ] RDS master password stored in Doppler or 1Password (NOT committed)

## Common issues

- **"InvalidClientTokenId"** — AWS credentials not loaded. `aws sts get-caller-identity` to verify
- **"AccessDenied" on RDS** — your IAM role needs `rds:*`, `kms:*`, `s3:*` permissions for bootstrap
- **KMS key deletion window warning** — 30 days is correct for prod; accept it
- **us-west-1 only has 2 AZs** — that's why subnets are in `us-west-1a` and `us-west-1c`, not `b`
